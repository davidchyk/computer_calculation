cmake_minimum_required(VERSION 3.20)

# Назва проєкту
project(MyCppApp LANGUAGES CXX)

# Встановити стандарт C++ (можна замінити на 20 або 23)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Шлях до виконуваного файлу — та сама директорія, що й .cpp
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR})

# Явно вказати використання MSVC (якщо важливо)
if (MSVC)
    message(STATUS "Компілятор: MSVC")
    add_compile_options(/W4 /permissive- /utf-8)
endif()

# Додати вихідний файл
add_executable(app
    main.cpp
    normal_forms.cpp
)
